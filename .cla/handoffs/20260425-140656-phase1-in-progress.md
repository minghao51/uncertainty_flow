```python
# tests/multivariate/test_copula.py (line ~117)
def test_fit_rejects_zero_variance_columns(self):
    """Should error when any target column has zero variance."""
    from uncertainty_flow.utils.exceptions import InvalidDataError
    residuals = np.column_stack([
        np.ones(100),  # Constant column (zero variance)
        np.random.randn(100) * 2 + 1,  # Variable column
    ])
    copula = GaussianCopula()
    with pytest.raises(InvalidDataError, match="zero variance"):
        copula.fit(residuals)

def test_fit_handles_high_correlation_with_conditioning(self):
    """Should handle high correlation via matrix conditioning."""
    np.random.seed(42)
    n_samples = 1000
    # Create bivariate residuals with correlation 0.99
    z = np.random.randn(n_samples)
    z[:, 1] = 0.99 * z[:, 0] + 0.14 * np.random.randn(n_samples)
    residuals = z.T

    copula = GaussianCopula()
    copula.fit(residuals)

    # Should succeed without error
    assert copula.fitted_

    # Sampling should work
    marginals = np.random.rand(1, 2, 11)
    samples = copula.sample(marginals, n_samples=100, random_state=42)
    assert samples.shape == (100, 2)
    assert np.all(np.isfinite(samples))
```
